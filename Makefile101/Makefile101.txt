1. 文本 / 文件名转换
subst：简单字符串替换
makefile
# 把 TEXT 里所有 foo 换成 bar
$(subst foo,bar,$(TEXT))
例子：

makefile
SRC := foo.c bar.c
OBJ := $(subst .c,.o,$(SRC))   # → foo.o bar.o

patsubst：按模式替换（带 %）
makefile
# 把所有 %.c 换成 %.o
$(patsubst %.c,%.o,$(SRC))
比 subst 更适合文件后缀这种场景。

2. 列表过滤 / 查找
filter：保留匹配的
makefile
# 只保留 .c 和 .s 文件
$(filter %.c %.s,$(FILES))

filter-out：丢掉匹配的
makefile
# 去掉 test_*.c
$(filter-out test_%.c,$(SRC))

findstring：是否包含子串
makefile
$(findstring foo,$(TEXT))  # 包含则返回 foo，不含则返回空

3. 单词操作（Make 把“用空格分开的东西”都当成“单词列表”）
word / words / wordlist
makefile
$(words $(SRC))           # SRC 里单词个数
$(word 2,$(SRC))          # 第二个单词
$(wordlist 2,4,$(SRC))    # 第2~4个单词
firstword
makefile
$(firstword $(SRC))       # 第一个单词，常用来取“主目标”

4. 文件名相关
dir / notdir
makefile
$(dir src/main.c lib/utils.c)      # → src/ lib/
$(notdir src/main.c lib/utils.c)   # → main.c utils.c

basename / suffix
makefile
$(basename main.c foo.cpp)   # → main foo
$(suffix main.c foo.cpp)     # → .c .cpp

addprefix / addsuffix
makefile
$(addprefix src/,$(FILES))   # foo.c → src/foo.c
$(addsuffix .o,$(NAMES))     # foo → foo.o
wildcard：按通配符找文件
makefile
SRC := $(wildcard src/*.c)   # 找出目录下所有 .c

5. 条件与循环
if
makefile
# 语法：$(if 条件,为真时的值,为假时的值)
DEBUG_FLAGS := $(if $(DEBUG),-g,-O2)

foreach
makefile
# 语法：$(foreach var,列表,表达式)
OBJS := $(foreach f,$(SRC),$(f:.c=.o))

call：自定义“函数”
makefile
define make-obj
$(1:.c=.o)
endef

OBJS := $(call make-obj,$(SRC))

6. 与外部命令交互
shell：执行一条 shell，取输出
makefile
GIT_HASH := $(shell git rev-parse --short HEAD)

makefile
$(origin CFLAGS)  # 命令行 / env / file / default 等
